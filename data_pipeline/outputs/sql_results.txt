================================================================================

            title,

            rating,
            in_stock
        FROM books

================================================================================
Query 6 - Books from selected categories
================================================================================

SQL:
SELECT
            title,
            category_id,
            rating
        FROM books
        WHERE category_id IN (
            SELECT category_id
            FROM categories
            WHERE category_name IN (
                'Academic',
                'Adult Fiction'
            )
        )
        ORDER BY rating DESC, title;

Output:
title | category_id | rating
----------------------------------------------------------------------------------------------------
Fifty Shades Freed (Fifty Shades #3) | 3 | 5
Logan Kade (Fallen Crest High #5.5) | 1 | 2


================================================================================
Query 7 - Books with category names using JOIN
================================================================================

SQL:
SELECT
            b.title,
            b.rating,
            b.price_gbp,
            b.price_inr,
            b.in_stock,
            c.category_name
        FROM books AS b
        INNER JOIN categories AS c
            ON b.category_id = c.category_id
        ORDER BY c.category_name, b.rating DESC, b.title
        LIMIT 20;
